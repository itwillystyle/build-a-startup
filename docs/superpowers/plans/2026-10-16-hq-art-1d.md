# HQ Part 1d: The Art. Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Dress the real tower (1a-1c) in its art: fins, banded floors with a lip and a light line, faceted round corners, a lobby entrance with the company's name, terraces with a planter edge and trees, a crown ring with the name plate, lift door frames, a tidy top on the shaft, a crane and rail on an unfinished floor, three finishes from one table, and a night look made of light lines instead of glowing glass.

**Architecture:** A pure `TowerFinish` module (ReplicatedStorage) holds the three finishes. `TowerPlan` gains the pure geometry the art needs (corners, arcs, the shaft's top, the unfinished floor, the rail edges, dark rooms). A new server module `TowerArt` builds the art from Parts through the two helpers `Wafers` injects (`slabBox`, `add`); `Wafers` calls it from the builders it already has. `SkyClient` switches parts tagged `SVNightLight` at night instead of turning facade glass to Neon. The picker's ViewportFrame reads `TowerFinish`.

**Tech Stack:** Roblox Luau, Rojo (files are the truth), Lune offline tests (`~/.rokit/bin/lune run tests/offline/run`), selene (`~/.rokit/bin/selene src`), Studio MCP for live checks (Wilz only), `lune run tools/bas checkup pr`.

**Spec:** `silicon-startup/docs/superpowers/specs/2026-10-16-hq-art-1d-design.md`. Mocks: `silicon-startup/ui/2026-10-16-art-1d/mocks/01..17`.

## Global Constraints
- Parts only, no new meshes or uploads (spec §2). Repeated part sizes so Roblox instances them.
- Colours on `Palette.ANCHORS` (`Palette.compliant` true); no pure white or black (CONSTRAINTS rule 12).
- Neon material only on light lines, ceiling strips, lamps and the beacon (rule 12); every one tagged `SVEmissive` at creation (SVStyle's loose-Neon check).
- No collision added where a player walks except the terrace planter edge, its rail and handrail, the hall roof and the shaft cap (spec §4.2). Everything else: `CanCollide = false, CanQuery = false, CanTouch = false, CastShadow = false`.
- The saved `hqPath` keys stay `W_` Glass, `T_` Timber, `D_` Neon; unknown/missing → Glass. No saved field changes.
- Performance: full tower (level 100) from the plaza ≤ 500k triangles and ≤ 400 draw calls (parent spec §8.3).
- CONSTRAINTS rule 10: no new top-level local in `SiliconCore.server.lua` (none planned). Rule 4: judge/test-rule changes in their own commits (none planned).
- Pure `ReplicatedStorage` modules require no sibling; offline tests load pure modules only.
- Studio tests as Wilz only; restore the save after every Play: `require(game.ServerScriptService.DevScenarios:Clone()).restore(1688749216)` 4 s after Stop (clean save: cash 1437329148, wl 55, placed 8).
- **Planning ruling (corner radius):** the spec asks 6 (office) / 12 (lobby). The furniture margin (`T.inner`: WALL_T + GLASS_BAND = 1.3) only fits a 3-panel corner up to radius 3.2 (worked in Task 2's test). The plan uses **`T.CORNER = 3`** everywhere; larger corners need furniture re-laid at the corners (project #2). Told to Luke at handoff.
- **Planning ruling (mast light):** spec says red; the palette has no red anchor. The beacon is BrandGold.
- **Planning ruling (cab sign):** "over the doors": it hangs inside the cab just under the floor display, over the door opening (bottom at 7.0, doors are 8 tall, avatars 5), facing the riders: the ride camera (1b) looks at the doors, so this is where riders see it.

```lua
-- ---------------------------------------------------------------- 1d (16 Oct): the art's geometry
--[[ Rounded corners: the building's outer corners close with three angled panels on an arc of
	radius T.CORNER. The radius is capped by the furniture margin (T.inner keeps furniture WALL_T +
	GLASS_BAND = 1.3 from a wall): a corner room's furniture must stay inside the panels (tower.spec),
	which holds up to 3.2. The mock's 8 needs furniture re-laid at the corners (project #2). ]]
T.CORNER = 3
T.LIP = 1.2   -- how far a floor band stands out past the glass

-- the building's outer corners in this room: { x, z, sx, sz } (the corner point, its outward signs)
function T.corners(storey, seg)
	local r = T.room(storey, seg)
	local s, out = r.sides, {}
	for _, c in ipairs({ { "px", "pz", r.x1, r.z1, 1, 1 }, { "px", "nz", r.x1, r.z0, 1, -1 },
		{ "nx", "pz", r.x0, r.z1, -1, 1 }, { "nx", "nz", r.x0, r.z0, -1, -1 } }) do
		if s[c[1]] == "glass" and s[c[2]] == "glass" then
			table.insert(out, { x = c[3], z = c[4], sx = c[5], sz = c[6] })
		end
	end
	return out
end

-- the 4 points of corner c's three panels, radius R, from the x-face to the z-face (inset > 0 pulls in)
function T.arc(c, R, inset)
	local cx, cz = c.x - c.sx * R, c.z - c.sz * R
	local rr = R - (inset or 0)
	local pts = {}
	for k = 0, 3 do
		local a = math.rad(k * 30)
		pts[k + 1] = { x = cx + c.sx * rr * math.cos(a), z = cz + c.sz * rr * math.sin(a) }
	end
	return pts
end

-- how far each glass side of this room stops short of a corner: side -> { lo, hi } (along x or z)
function T.cuts(storey, seg)
	local k = { px = { lo = 0, hi = 0 }, nx = { lo = 0, hi = 0 }, pz = { lo = 0, hi = 0 }, nz = { lo = 0, hi = 0 } }
	for _, c in ipairs(T.corners(storey, seg)) do
		local xs, zs = (c.sx > 0) and "px" or "nx", (c.sz > 0) and "pz" or "nz"
		if c.sz > 0 then k[xs].hi = T.CORNER else k[xs].lo = T.CORNER end   -- an x-face runs along z
		if c.sx > 0 then k[zs].hi = T.CORNER else k[zs].lo = T.CORNER end   -- a z-face runs along x
	end
	return k
end

local function aroundShaft(r)
	local sh = T.SHAFT
	return {
		{ x0 = r.x0, x1 = sh.x0, z0 = r.z0, z1 = r.z1 },
		{ x0 = sh.x1, x1 = r.x1, z0 = r.z0, z1 = r.z1 },
		{ x0 = sh.x0, x1 = sh.x1, z0 = sh.z1, z1 = r.z1 },
		{ x0 = sh.x0, x1 = sh.x1, z0 = r.z0, z1 = sh.z0 },
	}
end

-- the roof over storey s's centre hall, where nothing stands above it yet (the 1c "hole over the lobby")
function T.hallRoofRects(s) return aroundShaft(T.centre(s)) end

local function hasKind(pieces, level, kind)
	for L = 2, level do
		local pc = pieces[L]
		if pc and pc.kind == kind then return true end
	end
	return false
end

-- where the top of the shaft is (its cap stands here), or nil: no tower yet, or the lantern stands over it
function T.shaftCapY(pieces, level)
	if level < 2 or hasKind(pieces, level, "lantern") then return nil end
	local top = T.topStorey(pieces, level)
	return T.floorY(top) + T.storeyH(top) - T.SLAB
end

-- the top storey with some but not all of its 8 rooms: storey, { [seg] = true }; or nil
function T.unfinished(pieces, level)
	local built, top = {}, nil
	for L = 2, level do
		local pc = pieces[L]
		if pc and pc.kind == "segment" then
			built[pc.storey] = built[pc.storey] or {}
			built[pc.storey][pc.seg] = true
			if not top or pc.storey > top then top = pc.storey end
		end
	end
	if not top then return nil end
	local n = 0
	for _ in pairs(built[top]) do n += 1 end
	if n >= 8 then return nil end
	return top, built[top]
end

-- the safety rail on an unfinished office floor: the outer edges of its rooms not built yet, 0.6 in
function T.railEdges(pieces, level)
	local s, built = T.unfinished(pieces, level)
	local out = {}
	if not s or s == 0 then return out, s end
	local i = 0.6
	for seg = 1, 8 do
		if not built[seg] then
			local r = T.room(s, seg)

		end
	end
	return out, s
end
```

```lua
function TowerArt.nightLight(p, fin, opts)

end

function TowerArt.facade(model, plot, storey, seg, y, h, fin, glassProps)

	for side, kind in pairs(r.sides) do
		if kind == "glass" then
			local alongX = (side == "pz" or side == "nz")
			local lo = (alongX and r.x0 or r.z0) + cut[side].lo
			local hi = (alongX and r.x1 or r.z1) - cut[side].hi
			local face = (side == "pz" and r.z1) or (side == "nz" and r.z0) or (side == "px" and r.x1) or r.x0
			local o = (side == "pz" or side == "px") and 1 or -1
			local function rect(a0, a1, d0, d1, y0, y1, name, props)
				local f0, f1 = face + o * d0, face + o * d1
				if f0 > f1 then f0, f1 = f1, f0 end
				if alongX then return box(model, plot, name, a0, a1, f0, f1, y0, y1, props) end
				return box(model, plot, name, f0, f1, a0, a1, y0, y1, props)
			end
			if office then
				local n = math.max(1, math.floor((hi - lo) / fin.finStep + 0.5))
				for k = 0, n - 1 do   -- each room owns its low end: no two fins on one seam
					local a = lo + (hi - lo) * k / n
					rect(a - fin.finW / 2, a + fin.finW / 2, 0, fin.finD, y, top, "Fin", finP)
				end
				rect(lo, hi, 0, T.LIP, top, top + T.SLAB, "Band", bandP)
				if fin.planters and alongX then
					for a = lo + 4, hi - 4, 7 do
						rect(a - 2.4, a + 2.4, T.LIP - 0.2, T.LIP + 1.0, top + T.SLAB, top + T.SLAB + 0.9, "Planter", potP)
						rect(a - 2.2, a + 2.2, T.LIP - 0.1, T.LIP + 0.9, top + T.SLAB + 0.9, top + T.SLAB + 1.6, "Plants", leafP)
						rect(a - 1.8, a + 1.8, T.LIP + 0.5, T.LIP + 0.9, top - 1.8, top + T.SLAB, "Plants", leafP)
					end
				end
			end
			nightLight(rect(lo, hi, 0.2, T.LIP - 0.3, lineY0, lineY1, "LightLine", lineP), fin)
		end

	for _, cn in ipairs(T.corners(storey, seg)) do

			nightLight(panel(model, plot, "LightLine", line[k], line[k + 1], lineY0, lineY1, T.LIP - 0.5, lineP), fin)
		end
	end
end

end
```
